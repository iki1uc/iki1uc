// Pumpensystem – kein Zylinder, aber zylindrisches Verhalten export function PUMPEN_SYSTEM(input) { // Bogenverhalten statt Kolbenhub const curve = Math.sin(input.value / 100 * Math.PI); // Alt/Neu-Verhalten (historisch vs. modern) const mode = input.mode === "alt" ? 0.7 : 1.0; // Wette-Slide: gleitender Druckbogen const slide = curve * mode * input.value; return { curve, // Bogenform mode, // alt/neu slide, // resultierender Druckbogen usable: slide > 10 }; }
