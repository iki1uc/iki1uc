'rgba(100, 20, 50, 0.8)' : 'rgba(0, 80, 120, 0.6)'; let gradient = ctx.createRadialGradient(nucleus.x, nucleus.y, 0, nucleus.x, nucleus.y, 600); gradient.addColorStop(0, gradColor); gradient.addColorStop(1, 'rgba(0, 0, 0, 0)'); const sigmaZ = Math.sqrt(Math.max(0, varZ)); document.getElementById('focusZ').textContent = DEPTH.focusZ.toFixed(2); document.getElementById('spreadZ').textContent = sigmaZ.toFixed(2); document.getElementById('attackPower').textContent = NC.attackPower.toFixed(2); ctx.arc(nucleus.x, nucleus.y, radius, 0, Math.PI * 2); ctx.stroke(); } // Fokusring const focusRingR = 180 + DEPTH.focusZ * 60; ctx.strokeStyle = 'rgba(255, 102, 255, 0.35)'; ctx.lineWidth = 1.5; ctx.setLineDash([3, 6]);
