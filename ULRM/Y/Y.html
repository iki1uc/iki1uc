Y – Rotate

ID: ULRM-Y
ROLE: rotate
MODE: RAW / RESPO
STATUS: active

DATA:
  axis: r
  step: +1
  effect: "LAGE.r = LAGE.r + 1"
  modi: "Y"
  rawator: "spin"
  pimp: "Betrachtung: Rotation"

ENGINE:
  function ULRM_Y(LAGE){
      return {
          x: LAGE.x,
          y: LAGE.y,
          z: LAGE.z,
          r: (LAGE.r ?? 0) + 1,
          modi: "Y",
          role: "rotate"
      };
  }

RESULT:
  LAGE rotiert (r + 1).
