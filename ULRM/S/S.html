S – Rückwärts

ID: ULRM-S
ROLE: backward
MODE: RAW / RESPO
STATUS: active

DATA:
  axis: x
  step: -1
  effect: "LAGE.x = LAGE.x - 1"
  modi: "S"
  rawator: "reverse"
  pimp: "Betrachtung: Rückzug"

ENGINE:
  function ULRM_S(LAGE){
      return {
          x: LAGE.x - 1,
          y: LAGE.y,
          z: LAGE.z,
          modi: "S",
          role: "backward"
      };
  }

RESULT:
  LAGE bewegt sich rückwärts (x - 1).
