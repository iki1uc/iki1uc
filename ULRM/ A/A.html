ULRM-A · Links

ID: ULRM-A
ROLE: left

DATA:
  axis: y
  step: -1
  effect: "LAGE.y = LAGE.y - 1"
  modi: "A"

ENGINE:
  function ULRM_A(LAGE){
      return {
          x: LAGE.x,
          y: LAGE.y - 1,
          z: LAGE.z,
          modi: "A",
          role: "left"
      };
  }

RESULT:
  LAGE bewegt sich nach links (y - 1).
