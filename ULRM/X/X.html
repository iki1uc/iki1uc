ID: ULRM-X-SET
ROLE: neutral-matrix
MODE: RAW / RESPO / CORE
STATUS: active

  X   → neutral
  XX  → neutral-strong
  XXX → neutral-core

DIMENSION:
  Ebene: 0
  Funktion: Ruhe, Stabilität, Kernneutralität
  Bindung: ULRM → RAWATOR → STAGE → ADAPTIVE

  function ULRM_X_SET(LAGE){

          X:   { x:LAGE.x, y:LAGE.y, z:LAGE.z, modi:"X",   role:"neutral" },
          XX:  { x:LAGE.x, y:LAGE.y, z:LAGE.z, lock:true, modi:"XX",  role:"neutral-strong" },
          XXX: { x:LAGE.x, y:LAGE.y, z:LAGE.z, lock:true, core:true, modi:"XXX", role:"neutral-core" }

  }
